第二步：编写第一个同步程序

2

创建 sync_demo.py

"""
Ghost Hub - 第一个同步示例
这个程序演示了基本的记忆同步功能
"""
import asyncio
from ghost_channel_sdk import GhostChannelSDK, GhostChannelConfig


async def main():
    # 1. 创建 SDK 实例
    sdk = GhostChannelSDK(GhostChannelConfig(
        compression_level=9,      # 启用压缩
        audit_enabled=True,       # 启用审计
        completion_mode="apply",   # 等待 APPLIED ACK
    ))
    
    # 2. 定义初始状态
    old_state = {
        "version": "v1",
        "knowledge": {
            "user_preferences": {"theme": "dark", "language": "zh"}
        },
        "interactions": []
    }
    
    # 3. 定义更新后的状态
    new_state = {
        "version": "v2",
        "knowledge": {
            "user_preferences": {"theme": "dark", "language": "zh"}
        },
        "interactions": [
            {"type": "query", "content": "Hello"}
        ]
    }
    
    # 4. 执行同步
    result = await sdk.sync_memory_delta(
        source_role="agent_a",
        target_role="agent_b",
        old_state=old_state,
        new_state=new_state,
    )
    
    # 5. 查看结果
    print(f"同步成功: {result.success}")
    print(f"带宽降低: {result.bandwidth_reduction:.1%}")
    print(f"延迟: {result.latency_ms:.2f}ms")
    print(f"一致性验证: {result.consistency_verified}")


if __name__ == "__main__":
    asyncio.run(main())

第四步：体验工作流同步

4

创建工作流示例

"""
Ghost Hub - 工作流同步示例
演示如何同步具有依赖关系的工作流步骤
"""
import asyncio
from ghost_channel_sdk import GhostChannelSDK, GhostChannelConfig


async def workflow_demo():
    sdk = GhostChannelSDK(GhostChannelConfig())
    
    # 步骤1：获取依赖 (无依赖)
    result1 = await sdk.sync_workflow_state(
        workflow_id="build_pipeline",
        step_id="fetch_deps",
        step_state={"status": "completed", "packages": ["react", "redux"]},
        dependencies=[],
    )
    print(f"步骤1 完成: {result1.success}")
    
    # 步骤2：编译代码 (依赖步骤1)
    result2 = await sdk.sync_workflow_state(
        workflow_id="build_pipeline",
        step_id="compile",
        step_state={"status": "completed", "output": "/dist/app.js"},
        dependencies=["fetch_deps"],
    )
    print(f"步骤2 完成: {result2.success}")
    
    # 步骤3：运行测试 (依赖步骤2)
    result3 = await sdk.sync_workflow_state(
        workflow_id="build_pipeline",
        step_id="test",
        step_state={"status": "completed", "passed": 42},
        dependencies=["compile"],
    )
    print(f"步骤3 完成: {result3.success}")


asyncio.run(workflow_demo())

第五步：体验失败恢复

5

自动恢复机制

"""
Ghost Hub - 失败恢复示例
演示如何使用快照进行状态恢复
"""
import asyncio
from ghost_channel_sdk import GhostChannelSDK, GhostChannelConfig


async def recovery_demo():
    sdk = GhostChannelSDK(GhostChannelConfig())
    
    # 1. 创建工作流快照
    await sdk.sync_workflow_state(
        workflow_id="task_001",
        step_id="process",
        step_state={"status": "completed", "data": "important"},
        dependencies=[],
    )
    
    # 2. 模拟故障
    print("模拟故障发生...")
    
    # 3. 从快照恢复
    recovered = await sdk.recover_from_failure(
        step_id="process",
        last_known_state={"status": "failed"},
    )
    
    print(f"恢复状态: {recovered}")
    print(f"恢复后 data: {recovered.get('data')}")


asyncio.run(recovery_demo())
