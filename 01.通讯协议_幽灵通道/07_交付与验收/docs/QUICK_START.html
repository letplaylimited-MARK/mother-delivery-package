🚀 Ghost Hub SDK 快速开始

预计时间：5分钟 | 完成安装后立即体验

第一步：安装（30秒）

pip install ghost-hub-sdk

第二步：基础使用（2分钟）

2.1 初始化SDK

from ghost_hub_sdk import GhostHubSDK, GhostHubConfig

# 最简初始化
sdk = GhostHubSDK()

# 或自定义配置
config = GhostHubConfig(
    intention_bank_enabled=True,
    no_ui_adapter_enabled=True,
    agent_federation_enabled=True
)
sdk = GhostHubSDK(config)

2.2 执行工作流

# HR工作流示例
result = sdk.execute_workflow("帮我优化招聘流程")

print(f"工作流ID: {result['workflow_id']}")
print(f"任务数量: {result['task_graph']['task_count']}")
print(f"状态: {result['status']}")

2.3 意图匹配

# 自然语言匹配
match = sdk.intention_bank.match_intent("开灯")

print(f"匹配模板: {match.template_id}")
print(f"置信度: {match.confidence}")

第三步：IoT集成（1分钟）

3.1 连接设备

# 连接MQTT网关
sdk.no_ui_adapter.connect(protocol="mqtt")

3.2 发送命令

# 自然语言转设备命令
command = sdk.no_ui_adapter.convert_intent_to_command(
    intent="把灯打开",
    device_type="light"
)

# 发送到设备
result = sdk.no_ui_adapter.send_command(
    command=command,
    device_id="light_001"
)

第四步：智能体协作（1分钟）

# 创建任务
task = {
    "name": "数据分析",
    "description": "分析销售数据"
}

# 分发到智能体
result = sdk.agent_federation.distribute_task(
    task=task,
    strategy="capability_match"
)

print(f"分配给: {result['agent_id']}")

完整示例

#!/usr/bin/env python3
from ghost_hub_sdk import GhostHubSDK

def main():
    print("🚀 Ghost Hub SDK 快速开始\n")
    
    # 1. 初始化
    sdk = GhostHubSDK()
    
    # 2. 执行工作流
    result = sdk.execute_workflow("帮我优化招聘流程")
    print(f"✅ 创建了 {result['task_graph']['task_count']} 个任务")
    
    # 3. 意图匹配
    match = sdk.intention_bank.match_intent("开灯")
    print(f"✅ 匹配到: {match.template_id}")
    
    # 4. 清理
    sdk.disconnect()
    print("🎉 完成!")

if __name__ == "__main__":
    main()
