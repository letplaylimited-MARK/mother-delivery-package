🚀 2. SDK初始化

最简初始化

from ghost_hub_sdk import GhostHubSDK

sdk = GhostHubSDK()

自定义配置

from ghost_hub_sdk import GhostHubSDK, GhostHubConfig

config = GhostHubConfig(
    intention_bank_enabled=True,
    no_ui_adapter_enabled=True,
    agent_federation_enabled=True,
    log_level="INFO"
)

sdk = GhostHubSDK(config)

🎯 3. 意图银行

意图匹配

# 基本匹配
match = sdk.intention_bank.match_intent("帮我招聘工程师")

print(f"模板: {match.template_id}")
print(f"置信度: {match.confidence:.2f}")

领域过滤

# 仅HR领域
hr_match = sdk.intention_bank.match_intent(
    "绩效考核",
    domain="hr"
)

# 仅IoT领域
iot_match = sdk.intention_bank.match_intent(
    "开灯",
    domain="iot"
)
💡 提示：指定领域可以提高匹配准确度！

🏠 4. IoT适配器

连接设备

# MQTT连接
sdk.no_ui_adapter.connect(protocol="mqtt", broker="mqtt://localhost:1883")

# HTTP连接
sdk.no_ui_adapter.connect(protocol="http", base_url="http://localhost:8080")

发送命令

# 自然语言转命令
command = sdk.no_ui_adapter.convert_intent_to_command(
    intent="把客厅灯调暗",
    device_type="light"
)

# 发送到设备
result = sdk.no_ui_adapter.send_command(
    command=command,
    device_id="light_living_room"
)

场景执行

# 执行预定义场景
results = sdk.no_ui_adapter.execute_scene("离家模式")
print(f"执行了 {len(results)} 个设备命令")

🤖 5. 智能体联邦

任务分发

# 创建任务
task = {
    "name": "销售数据分析",
    "description": "分析Q1销售数据",
    "priority": "high"
}

# 分发任务
result = sdk.agent_federation.distribute_task(
    task=task,
    strategy="capability_match"
)

print(f"分配给: {result['agent_id']}")

协作会话

# 创建会话
session = sdk.agent_federation.create_session(
    session_id="project_alpha",
    agents=["data_collector", "analyzer", "reporter"]
)

session.start()
session.wait()
final_result = session.get_result()

⭐ 6. 最佳实践

⚠️ 错误处理

try:
    result = sdk.execute_workflow("招聘流程")
except GhostHubError as e:
    print(f"SDK错误: {e}")
finally:
    sdk.disconnect()  # 始终清理

✅ 性能优化

使用批量操作替代循环单个命令：

# ❌ 低效
for device in devices:
    sdk.no_ui_adapter.send_command(...)

# ✅ 高效
commands = [sdk.no_ui_adapter.convert_intent_to_command(...) for device in devices]
results = sdk.no_ui_adapter.send_batch_commands(commands)
